# M8b Usage Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Usage screen (Cost tab with stacked daily bars, threshold line and by-agent / by-model tables; Tools tab) and a daily cost alert (Settings → Alerts, Dashboard COST row, desktop notification).

**Architecture:** Two read-only collector queries (`usage_summary`, `usage_tools`) over the existing `--query` channel bucket sessions by the user's local day boundaries sent from the renderer or main. A `CostAlertService` in the main process re-runs `usage_summary` for today every 5 minutes and notifies once per local day. The renderer gets two stores, an IO module, pure libs, the screen and a Settings tab.

**Tech Stack:** Python 3 stdlib (collector, pytest), TypeScript, zod, Electron IPC and `Notification`, React 19, zustand, SVG, Vitest, Playwright for Electron.

**Spec:** `docs/superpowers/specs/2026-10-13-m8b-usage-design.md`

## Global Constraints

- UI strings are English only; replies to the user are Indonesian.
- Collector stays stdlib-only and **ASCII-only**.
- Every `state.db` is opened with `open_ro` / `open_profile`; only `messages.active = 1` rows are read; only counts, sums and the first 400 characters of tool results leave the VPS.
- The renderer and main send numbers and profile names only; `ipc.ts` validates with zod (profiles by `ProfileName`, at most 31 `dayStarts`, ascending, 82800 to 90000 seconds apart; `endTs` after the last start).
- Nothing is written on the VPS in this milestone.
- Day boundaries are local midnights of the user's machine, computed with `new Date(y, m, d)`.
- Shared range values: `today | 7d | 14d | 30d | all`; Usage windows: 7, 14, 30 days (`today` counts as 7, `all` as 30, without rewriting the setting).
- Cost threshold: `alerts.cost.usd` 0.01 to 100000 (default 10), `alerts.cost.enabled` default false.
- Heredoc gotcha on this machine: write files with the Write/Edit tools (a double backslash collapses in Bash heredocs).
- Run Python tests with `python`; Vitest with `npm test`; typecheck with `npm run typecheck`; e2e with `npm run build` then `npx playwright test <file>`.

import type { Range } from './sessions'

export interface UsageSummaryArgs {
  /** Empty means every profile. */
  profiles: string[]
  /** Local midnights in epoch seconds, ascending. */
  dayStarts: number[]
  /** The midnight after the last day. */
  endTs: number
}

export interface UsageToolsArgs {
  profiles: string[]
  sinceTs: number
  endTs: number
}

export const COST_MIN = 0.01
export const COST_MAX = 100000
export const COST_DEFAULT = 10

export const validCostThreshold = (n: unknown): n is number => typeof n === 'number' && Number.isFinite(n) && n >= COST_MIN && n <= COST_MAX

/** Usage shows 7, 14 or 30 days. `today` counts as 7 and `all` as 30; the setting itself is never rewritten. */
export function usageDays(range: Range): 7 | 14 | 30 {
  return range === '14d' ? 14 : range === '30d' || range === 'all' ? 30 : 7
}

/** Local midnights of the last `days` days (today last) and the midnight after today. DST safe: each one is built from a calendar date. */
export function dayStartsFor(days: number, now: number): { dayStarts: number[]; endTs: number } {
  const t = new Date(now * 1000)
  const at = (offset: number): number => new Date(t.getFullYear(), t.getMonth(), t.getDate() + offset).getTime() / 1000
  const dayStarts: number[] = []
  for (let i = days - 1; i >= 0; i--) dayStarts.push(at(-i))
  return { dayStarts, endTs: at(1) }
}

const p2 = (n: number): string => String(n).padStart(2, '0')

export function localDayKey(ts: number): string {
  const d = new Date(ts * 1000)
  return `${d.getFullYear()}-${p2(d.getMonth() + 1)}-${p2(d.getDate())}`
}

# ---- usage (M8b) ------------------------------------------------------------

USAGE_COLS = ["model", "billing_provider", "input_tokens", "output_tokens", "cache_read_tokens", "api_call_count",
              "estimated_cost_usd", "actual_cost_usd", "cost_status"]
USAGE_SESSIONS_MAX = 50000
USAGE_ROWS_MAX = 5000
USAGE_DAYS_MAX = 31


def _usage_cost(r):
    """(cost, estimated part, unknown flag) for one session, by the same rules as the session list."""
    if r.get("cost_status") not in (None, "", "actual", "estimated"):
        return 0.0, 0.0, 1
    actual, est = _opt_num(r.get("actual_cost_usd")), _opt_num(r.get("estimated_cost_usd"))
    if actual is not None:
        return actual, 0.0, 0
    if est is not None:
        return est, est, 0
    return 0.0, 0.0, 0


def _day_starts(a):
    starts = a.get("dayStarts")
    if not isinstance(starts, list) or not 1 <= len(starts) <= USAGE_DAYS_MAX:
        raise QueryError("bad_args")
    for v in starts:
        if isinstance(v, bool) or not isinstance(v, (int, float)):
            raise QueryError("bad_args")
    if any(b <= a_ for a_, b in zip(starts, starts[1:])):
        raise QueryError("bad_args")
    end = a.get("endTs")
    if isinstance(end, bool) or not isinstance(end, (int, float)) or end <= starts[-1]:
        raise QueryError("bad_args")
    return starts, end


def _select_cols(scols, wanted):
    return ", ".join(["s.started_at AS started_at"] + [("s.%s AS %s" % (c, c)) if c in scols else ("NULL AS %s" % c) for c in wanted])


def q_usage_summary(home, a):
    profiles = _filters({"profiles": a.get("profiles", [])})[0]
    starts, end = _day_starts(a)
    groups, errors, readable, truncated = {}, [], 0, False
    for name, db in _each_profile(home, profiles):
        con = None
        try:
            con = open_ro(db)
            scols = table_columns(con, "sessions")
            if not _has_sessions(scols):
                errors.append({"profile": name, "code": "unsupported_schema"})
                continue
            sql = "SELECT %s FROM sessions s WHERE s.started_at >= ? AND s.started_at < ? ORDER BY s.started_at DESC LIMIT ?" % _select_cols(scols, USAGE_COLS)
            got = _dict_rows(con.execute(sql, (starts[0], end, USAGE_SESSIONS_MAX + 1)))
            if len(got) > USAGE_SESSIONS_MAX:
                truncated = True
                got = got[:USAGE_SESSIONS_MAX]
            for r in got:
                ts = to_epoch(r.get("started_at"))
                if ts is None:
                    continue
                day = bisect.bisect_right(starts, ts) - 1
                if day < 0:
                    continue
                k = (day, name, _text(r.get("billing_provider"), 64), _text(r.get("model"), 128))
                g = groups.setdefault(k, {"sessions": 0, "inputTokens": 0, "outputTokens": 0, "cacheReadTokens": 0, "apiCalls": None, "cost": 0.0, "estimatedCost": 0.0, "unknownCost": 0})
                cost, est, unknown = _usage_cost(r)
                g["sessions"] += 1
                g["inputTokens"] += _opt_int(r.get("input_tokens")) or 0
                g["outputTokens"] += _opt_int(r.get("output_tokens")) or 0
                g["cacheReadTokens"] += _opt_int(r.get("cache_read_tokens")) or 0
                calls = _opt_int(r.get("api_call_count"))
                if calls is not None:
                    g["apiCalls"] = (g["apiCalls"] or 0) + calls
                g["cost"] += cost
                g["estimatedCost"] += est
                g["unknownCost"] += unknown
            readable += 1
        except sqlite3.Error:
            errors.append({"profile": name, "code": "db_unreadable"})
        finally:

import type { CostAlertState, UsageSummary } from '@shared/events'
import type { QueryResult } from '@shared/ipc'
import { dayStartsFor, localDayKey, type UsageSummaryArgs } from '@shared/usage'

export function evaluateCost(total: number, enabled: boolean, usd: number, notifiedDay: string | null, day: string): { over: boolean; notify: boolean } {
  const over = enabled && total > usd
  return { over, notify: over && notifiedDay !== day }
}

export interface CostAlertDeps {
  query(args: UsageSummaryArgs): Promise<QueryResult<UsageSummary>>
  settings(): { enabled: boolean; usd: number }
  isConnected(): boolean
  send(s: CostAlertState): void
  notify(s: CostAlertState): void
  now?: () => number
  intervalMs?: number
}

const FIVE_MINUTES = 5 * 60_000

/** Checks today's cost of all agents against the daily threshold while connected; one notification per local day. */
export class CostAlertService {
  private timer: ReturnType<typeof setInterval> | null = null
  private notifiedDay: string | null = null
  private running = false

  constructor(private readonly d: CostAlertDeps) {}

  start(): void {
    this.stop()
    this.timer = setInterval(() => void this.recheck(), this.d.intervalMs ?? FIVE_MINUTES)
    void this.recheck()
  }

  stop(): void {
    if (this.timer) clearInterval(this.timer)
    this.timer = null
  }

  reset(): void {
    this.notifiedDay = null
  }

  async recheck(): Promise<void> {
    if (this.running) return
    this.running = true
    try {
      const nowMs = (this.d.now ?? Date.now)()
      const day = localDayKey(nowMs / 1000)
      const { enabled, usd } = this.d.settings()
      if (!enabled) {
        this.d.send({ enabled: false, day, total: null, threshold: usd, over: false, checkedAt: nowMs })
        return
      }
      if (!this.d.isConnected()) return
      const { dayStarts, endTs } = dayStartsFor(1, nowMs / 1000)
      const res = await this.d.query({ profiles: [], dayStarts, endTs })
      if (!res.ok) return
      const total = res.data.rows.reduce((sum, r) => sum + r.cost, 0)
      const { over, notify } = evaluateCost(total, enabled, usd, this.notifiedDay, day)
      const state: CostAlertState = { enabled, day, total, threshold: usd, over, checkedAt: nowMs }
      this.d.send(state)
      if (notify) {
        this.notifiedDay = day
        this.d.notify(state)
      }
    } finally {
      this.running = false
    }
  }
}

export function niceTicks(max: number, count = 4): { max: number; ticks: number[] } {
  const top = Math.max(max, 0.01)
  const rough = top / count
  const pow = Math.pow(10, Math.floor(Math.log10(rough)))
  const frac = rough / pow
  const step = (frac <= 1 ? 1 : frac <= 2 ? 2 : frac <= 2.5 ? 2.5 : frac <= 5 ? 5 : 10) * pow
  const n = Math.max(1, Math.ceil(top / step - 1e-9))
  const ticks = Array.from({ length: n + 1 }, (_, i) => Math.round(i * step * 1e6) / 1e6)
  return { max: ticks[ticks.length - 1]!, ticks }
}
